import java.util.ArrayList; import java.util.Collections; import java.util.List; import java.util.Scanner; public class CroupierVirtuel { // --- RÈGLE DU CASINO : SOFT 17 --- // Mettre à 'true' si vous voulez que le croupier tire aussi sur un 17 "doux" (ex: As + 6) private static final boolean TIRER_SUR_SOFT_17 = false; public static void main(String[] args) { Scanner scanner = new Scanner(System.in); Paquet paquet = new Paquet(); System.out.println("=========================================="); System.out.println(" CROUPIER VIRTUEL 100% AUTOMATIQUE"); System.out.println("=========================================="); while (true) { System.out.println("\n--- NOUVELLE MANCHE ---"); System.out.println("Appuyez sur ENTRÉE pour lancer la distribution du croupier (ou tapez 'quitter')..."); String input = scanner.nextLine(); if (input.equalsIgnoreCase("quitter")) break; JouerMancheCroupier(paquet); } scanner.close(); } private static void JouerMancheCroupier(Paquet paquet) { List mainCroupier = new ArrayList<>(); // 1. Le croupier tire ses 2 cartes initiales mainCroupier.add(paquet.tirerCarte()); mainCroupier.add(paquet.tirerCarte()); // 2. Affichage de la première carte (la 2e reste cachée pour l'instant) System.out.println("\n🂠 Carte visible du croupier : [ " + mainCroupier.get(0) + " ]"); System.out.println("🂠 Seconde carte : [ ❓ Cachée ]"); // 3. Pause pendant que les vrais joueurs jouent sur la table System.out.println("\n➡ Jouez vos tours autour de la table !"); System.out.println("➡ Une fois tous les joueurs servis, appuyez sur ENTRÉE pour révéler la main du croupier."); new Scanner(System.in).nextLine(); // 4. Révélation de la seconde carte System.out.println("--- RÉVÉLATION DU CROUPIER ---"); System.out.println("Cartes en main : " + affichageMain(mainCroupier)); System.out.println("Score initial : " + calculerScore(mainCroupier)); // 5. Tour automatique du croupier while (devraitTirer(mainCroupier)) { System.out.println("\n👉 Le croupier doit tirer une carte..."); attendre(1500); // Petite pause pour le suspense Carte nouvelleCarte = paquet.tirerCarte(); mainCroupier.add(nouvelleCarte); System.out.println("Carte tirée : [ " + nouvelleCarte + " ]"); System.out.println("Main actuelle : " + affichageMain(mainCroupier)); System.out.println("Nouveau score : " + calculerScore(mainCroupier)); } // 6. Fin de manche int scoreFinal = calculerScore(mainCroupier); System.out.println("\n------------------------------------------"); if (scoreFinal > 21) { System.out.println("💥 LE CROUPIER BRÛLE (BUST) avec " + scoreFinal + " !"); System.out.println("✅ Tous les joueurs encore en lice gagnent."); } else { System.out.println("✋ LE CROUPIER S'ARRÊTE À " + scoreFinal + "."); System.out.println("📊 Comparez vos scores : battez " + scoreFinal + " pour gagner !"); } System.out.println("------------------------------------------"); } // Logique de décision du croupier (Règle stricte des 17) private static boolean devraitTirer(List main) { int score = calculerScore(main); if (score < 17) return true; // Option Soft 17 : Si activée, le croupier tire sur un 17 comprenant un As valant 11 if (score == 17 && TIRER_SUR_SOFT_17 && estSoft(main)) { return true; } return false; } // Calcul dynamique du score avec gestion de l'As (11 ou 1) private static int calculerScore(List main) { int total = 0; int nombreDeAs = 0; for (Carte c : main) { total += c.valeur; if (c.valeur == 11) nombreDeAs++; } while (total > 21 && nombreDeAs > 0) { total -= 10; nombreDeAs--; } return total; } private static boolean estSoft(List main) { int total = 0; int nombreDeAs = 0; for (Carte c : main) { total += c.valeur; if (c.valeur == 11) nombreDeAs++; } return (total == 17 && nombreDeAs > 0); } private static String affichageMain(List main) { StringBuilder sb = new StringBuilder(); for (Carte c : main) { sb.append("[").append(c).append("] "); } return sb.toString().trim(); } private static void attendre(int ms) { try { Thread.sleep(ms); } catch (InterruptedException ignored) {} } // --- STRUCTURE DES CARTES & PAQUET DE 52 CARTES --- static class Carte { String nom; int valeur; Carte(String nom, int valeur) { this.nom = nom; this.valeur = valeur; } @Override public String toString() { return nom; } } static class Paquet { private final List cartes = new ArrayList<>(); Paquet() { reinitialiser(); } void reinitialiser() { cartes.clear(); String[] couleurs = {"♥️", "♦️", "♣️", "♠️"}; String[] noms = {"2", "3", "4", "5", "6", "7", "8", "9", "10", "Valet", "Dame", "Roi", "As"}; int[] valeurs = {2, 3, 4, 5, 6, 7, 8, 9, 10, 10, 10, 10, 11}; for (String couleur : couleurs) { for (int i = 0; i < noms.length; i++) { cartes.add(new Carte(noms[i] + " " + couleur, valeurs[i])); } } Collections.shuffle(cartes); } Carte tirerCarte() { if (cartes.size() < 10) { // Remélange si le paquet est presque vide System.out.println("\n🔄 [Mélange du paquet de cartes...]"); reinitialiser(); } return cartes.remove(0); } } }
